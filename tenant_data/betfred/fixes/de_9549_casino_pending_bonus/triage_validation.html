Proposed Glue expression

LEFT JOIN sgp_currency ccy
  ON UPPER(TRIM(ccy.isocode)) = UPPER(TRIM(a.currency_code))
LEFT JOIN sgp_currency_exchange_rate fx
  ON fx.currency_bkfrom = ccy.currency_bk
 AND fx.currency_bkto = 4
 AND fx.date = CAST(a.action_date AS DATE)   -- or bonus_start_date

CASE
  WHEN UPPER(TRIM(a.currency_code)) = 'GBP'
    THEN CAST(a.amount AS DECIMAL(19, 8))
  WHEN fx.divisor IS NOT NULL AND fx.divisor <> 0
    THEN CAST(a.amount * fx.multiplier / fx.divisor AS DECIMAL(19, 8))
  ELSE NULL
END AS amount_gbp
